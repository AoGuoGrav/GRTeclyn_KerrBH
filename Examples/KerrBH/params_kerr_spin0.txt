#################################################
# Kerr initial-data parameters

# M is the black-hole mass.
kerr.mass = 1.0

# Kerr parameter a = J / M.
# This file tests the Schwarzschild limit first.
kerr.spin = 0.0

# spin_direction is physically irrelevant for spin = 0, but it must be
# non-zero because KerrBHInitialData validates it.
kerr.spin_direction = 0.0 0.0 1.0

# Keep the black hole at the centre of the full computational domain.
# geometry.center is used as the fallback; kerr.center explicitly matches it.

kerr.offset = 0.0 0.0 0.0

#################################################
# Grid parameters
#
# Use a full 3D domain now, so this exact file can later test a spin

geometry.prob_extent = 8.0 8.0 8.0

# There is one level only during the initial-data validation.
amr.max_level = 0

amr.max_grid_size = 16

#################################################
# Boundary conditions
#
# Do not use a reflective z boundary. It would be invalid for a general

boundary.hi_condition = "SOMMERFELD_BC" "SOMMERFELD_BC" "SOMMERFELD_BC"
boundary.lo_condition = "SOMMERFELD_BC" "SOMMERFELD_BC" "SOMMERFELD_BC"

#################################################
