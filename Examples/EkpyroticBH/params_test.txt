# GRTeclyn: black hole in an ekpyrotic contraction, XCTS initial data
# Small single-level test of the XCTS relaxation (see XCTSRelaxation.hpp and
# EkpyroticBHData.hpp). Commented parameters show default or suggested
# values.
#
# Checks:
#  - bh.mass = 0 xcts.lapse = unit must reproduce the Ekpyrotic example
#    (beta stays 0, lapse stays 1).
#  - bh.mass > 0: the XCTS residuals Ru, Rv, Rbeta printed every step decay.

#################################################

#################################################

#################################################
# Black hole parameters

#################################################

# derivatives) evolve; the defaults of A_* and eta_* are printed at start-up.

xcts.relaxation_time = 5.0
# "xcts" solves the lapse equation, "unit" sets alpha = 1 (CTS)
xcts.lapse = xcts
